cmake_minimum_required(VERSION 3.24)
project(sample VERSION 0.1.0 LANGUAGES C CXX)
include(@PLATFORM@/cxx/cmake/house.cmake)
house_cxx_common_settings(OFF)
house_cxx_add_strict_target(sample_strict LANGUAGES CXX)
enable_testing()
add_subdirectory(src/value)
add_executable(sample_test tests/main.cpp)
target_link_libraries(sample_test PRIVATE sample_value sample_strict)
add_test(NAME sample_value COMMAND sample_test)
add_executable(sample_consumer_test tests/consumer/main.cpp)
target_link_libraries(sample_consumer_test PRIVATE sample_value sample_strict)
add_test(NAME sample_consumer COMMAND sample_consumer_test)
house_cxx_install_package(PACKAGE sample VERSION ${PROJECT_VERSION}
    NAMESPACE sample TARGETS sample_value sample_strict)
install(DIRECTORY include/ DESTINATION include)
